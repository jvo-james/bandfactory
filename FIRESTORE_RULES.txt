rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function isAdmin() {
      return signedIn()
        && exists(/databases/$(database)/documents/admins/$(request.auth.uid))
        && get(/databases/$(database)/documents/admins/$(request.auth.uid)).data.active == true;
    }

    match /admins/{uid} {
      allow read: if signedIn() && request.auth.uid == uid;
      allow write: if false;
    }

    match /settings/{doc} {
      allow read: if true;
      allow write: if isAdmin();
    }

    match /products/{doc} {
      allow read: if true;
      // Admins may manage the full product record. During checkout, the verified-payment
      // flow may only change inventory styles and the updatedAt timestamp.
      allow create, delete: if isAdmin();
      allow update: if isAdmin();
    }

    match /reviews/{id} {
      allow read: if isAdmin() || resource.data.status == 'approved';
      allow create: if request.resource.data.status == 'pending';
      allow update, delete: if isAdmin();
    }

    match /subscribers/{id} {
      allow create: if true;
      allow read, update, delete: if isAdmin();
    }

    match /messages/{id} {
      allow create: if true;
      allow read, update, delete: if isAdmin();
    }

    match /orders/{id} {
      allow create: if true;
      allow read, update, delete: if isAdmin();
    }

    match /customers/{id} {
      allow create: if true;
      allow read, update, delete: if isAdmin();
    }

    match /notifications/{id} {
      allow create: if true;
      allow read, update, delete: if isAdmin();
    }

    match /abandonedCarts/{id} {
      allow create, update: if true;
      allow get: if true;
      allow list, delete: if isAdmin();
    }

    match /activity/{id} {
      allow create: if true;
      allow read, update, delete: if isAdmin();
    }

    match /paymentReferences/{id} {
      allow create: if true;
      allow read, update, delete: if isAdmin();
    }
  }
}
